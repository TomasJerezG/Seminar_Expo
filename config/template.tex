
% Configuraciones generales de Beamer, la template y otros...

% Quita los controles default de Beamer
\setbeamertemplate{navigation symbols}{}

% --- Colores beamer ---
\setbeamercolor{normal text}      {fg=unalDk1}
\setbeamercolor{background canvas}{bg=white}
\setbeamercolor{itemize item}     {fg=unalAcc5}
\setbeamercolor{enumerate item}   {fg=unalAcc5}

\setbeamercolor{headerline}       {bg=unalAcc2}
\setbeamercolor{footerline}       {bg=unalAcc4}

\setbeamercolor{footlinecolor}    {fg=unalBg2,bg=unalDk2}

\setbeamertemplate{itemize item}  {\textcolor{unalAcc5}{\textbullet}}
\setbeamertemplate{enumerate item}{\textcolor{unalAcc5}{\insertenumlabel.}}

% --- Headline (base) ---
% Los espacios de margen de las lineas son aprox. 3.5% horizontalmente y 5% verticalmente.
% El tamaño del 'bloque' al final, y inicio en el footline, de la línea aprox. 19% horizontalmente y 1.5% verticalmente.
% Según las imágenes de 'referencia' de la plantilla
\setbeamertemplate{headline}
{%
    \vspace*{0.025\paperheight}
    \fontsize{8pt}{12pt}
    \color{unalAcc1}
    \hspace*{0.035\paperwidth}\makebox[\textwidth][l]{\AncizarSansLight{\insertframenumber{}}}
    \TMPTheaderline{}
}

% HeaderLine y FooterLine
% Decoraciones del header y footer en las slides.
\NewDocumentCommand{\TMPTheaderline}{}
{%
    \hspace*{0.035\paperwidth}{\color{unalAcc2} \rule[-0.02\paperheight]{0.93\paperwidth}{0.7pt}}
    \hspace*{0.775\paperwidth}{\color{unalAcc2} \rule{0.19\paperwidth}{0.014\paperheight}}
}
% Me toco poner una linea que llenada el espacio del bloque del inicio. COMO ME ENCANTA LaTex.
\NewDocumentCommand{\TMPTfooterline}{}
{%
    \hspace*{0.035\paperwidth}{\color{unalAcc4} \rule{0.19\paperwidth}{0.014\paperheight}}
    {\color{unalAcc1} \rule[2pt]{0.74\paperwidth}{0.7pt}}
    \hspace*{0.035\paperwidth}{\color{unalAcc4} \rule{0.93\paperwidth}{0.7pt}}
}

% --- Footline ---
% El tamaño del pie de linea aprox. 10.8% de altura.
% Según las imágenes de 'referencia' de la plantilla
\setbeamertemplate{footline}
{%
    \begin{beamercolorbox}[wd=\paperwidth,ht=0.08\paperheight]{footlinecolor}
        \begin{columns}
            \begin{column}{0.7\paperwidth}
                \fontsize{8pt}{8.75pt}
                \color{unalBg2}%
                \AncizarSansItalic{Facultad de \Facultad} \\
                \AncizarSansItalic{Sede \Sede}
                \vspace*{5pt}
            \end{column}
            \begin{column}{0.1\paperwidth}
                bye
            \end{column}
        \end{columns}
    \end{beamercolorbox}
}

% --- Title / Subtitle ---
\setbeamertemplate{frametitle}
{%
    \fontsize{18pt}{20pt}%
    \color{unalAcc1}%
    \AncizarSerifEB{\insertframetitle{}} \\
    \fontsize{10pt}{12pt}%
    \AncizarSansLight{\MakeUppercase{\insertframesubtitle{}}}
}
